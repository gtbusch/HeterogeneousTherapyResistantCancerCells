## repeat the following steps for each sample

#make a folder for the alignments

mkdir -p alignments

#increase the number of tasks that can be processed

ulimit -n 4096

#align the fasts with hisat2 (make sure reference genome is 

hisat2 -x ~/hisat2_grch38 \
       -1 RC_Sample1_merged_R1.fastq \
       -2 RC_Sample1_merged_R2.fastq \
       -p 8 \
       --summary-file RC_Sample1_alignment_summary.txt \
       | samtools view -bS - | samtools sort -o alignments/RC_Sample1.bam

#index alignments

samtools index alignments/RC_Sample1.bam

#check to make sure the file was generated, check stats

ls -lh alignments/
samtools flagstat alignments/RC_Sample1.bam
